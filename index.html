// --- 1. 原有基礎：初始化 LIFF --- async function initLiff() { await liff.init({ liffId: "2009620784-76eBZ9C0" }); if (liff.isLoggedIn()) { const profile = await liff.getProfile(); const uid = profile.userId; sessionStorage.setItem('lineUid', uid); // 存起來備用 // --- 2. 核心邏輯：檢查綁定狀態 --- checkBinding(uid); } else { // 未登入顯示登入按鈕 document.getElementById('loginDiv').style.display = 'block'; } } // --- 3. 檢查是否綁定社群名稱 --- function checkBinding(uid) { const api = "你的_GAS_網址"; fetch(`${api}?uid=${uid}`) .then(res => res.json()) .then(data => { if (data.status === "success") { // 已綁定：隱藏登入區，顯示歡迎詞與商品清單 document.getElementById('userName').innerText = data.name + "，歡迎回來 🦋"; document.getElementById('shopArea').style.display = 'block'; loadProducts(); // <--- 關鍵：直接載入商品 } else { // 未綁定：導向綁定頁面 (check.html) window.location.href = "check.html"; } }); } // --- 4. 載入並渲染商品清單 --- function loadProducts() { const api = "你的_GAS_網址?action=getProducts"; fetch(api) .then(res => res.json()) .then(products => { const grid = document.getElementById('productGrid'); grid.innerHTML = products.map(p => `

${p.name}

NT$ ${p.price}

${p.tag || ''}
`).join(''); }); }
